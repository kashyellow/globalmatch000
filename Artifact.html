// components/DestinationSearch.tsx - Search bar for 42 destinations
// Drop into your homepage header where it says "LIVE IN 42 DESTINATIONS"

"use client"
import { useState, useMemo } from "react"
import { useRouter } from "next/navigation"

const DESTINATIONS = [
  { country: "Dominican Republic", city: "Encuentro, Sosua DR", slug: "encuentro-sosua-dr", flag: "🇩🇴" },
  { country: "Indonesia", city: "Ubud, Bali", slug: "ubud-bali", flag: "🇮🇩" },
  { country: "Indonesia", city: "Canggu, Bali", slug: "canggu-bali", flag: "🇮🇩" },
  { country: "Colombia", city: "Medellin", slug: "medellin-colombia", flag: "🇨🇴" },
  { country: "Portugal", city: "Lisbon", slug: "lisbon-portugal", flag: "🇵🇹" },
  { country: "Mexico", city: "Mexico City", slug: "mexico-city-mexico", flag: "🇲🇽" },
  { country: "Mexico", city: "Tulum", slug: "tulum-mexico", flag: "🇲🇽" },
  { country: "Thailand", city: "Chiang Mai", slug: "chiang-mai-thailand", flag: "🇹🇭" },
  { country: "Thailand", city: "Bangkok", slug: "bangkok-thailand", flag: "🇹🇭" },
  { country: "Argentina", city: "Buenos Aires", slug: "buenos-aires-argentina", flag: "🇦🇷" },
  { country: "Brazil", city: "Rio de Janeiro", slug: "rio-brazil", flag: "🇧🇷" },
  { country: "South Africa", city: "Cape Town", slug: "cape-town-sa", flag: "🇿🇦" },
  { country: "Morocco", city: "Marrakech", slug: "marrakech-morocco", flag: "🇲🇦" },
  { country: "Spain", city: "Barcelona", slug: "barcelona-spain", flag: "🇪🇸" },
  { country: "Germany", city: "Berlin", slug: "berlin-germany", flag: "🇩🇪" },
  { country: "Czech Republic", city: "Prague", slug: "prague-czech", flag: "🇨🇿" },
  { country: "Vietnam", city: "Hoi An", slug: "hoi-an-vietnam", flag: "🇻🇳" },
  { country: "Vietnam", city: "Da Nang", slug: "da-nang-vietnam", flag: "🇻🇳" },
  { country: "Japan", city: "Kyoto", slug: "kyoto-japan", flag: "🇯🇵" },
  { country: "Peru", city: "Lima", slug: "lima-peru", flag: "🇵🇪" },
  // Add remaining to reach 42...
]

export default function DestinationSearch() {
  const [query, setQuery] = useState("")
  const [isOpen, setIsOpen] = useState(false)
  const router = useRouter()

  const filtered = useMemo(() => {
    if (!query) return DESTINATIONS.slice(0, 8) // show top 8 when empty
    const q = query.toLowerCase()
    return DESTINATIONS.filter(d => 
      d.country.toLowerCase().includes(q) || 
      d.city.toLowerCase().includes(q) ||
      d.slug.includes(q)
    ).slice(0, 10)
  }, [query])

  const goTo = (slug: string) => {
    router.push(`/hub/${slug}`)
    setIsOpen(false)
    setQuery("")
  }

  return (
    <div className="relative w-full max-w-[520px] mx-auto">
      {/* Search Input */}
      <div className="relative">
        <div className="absolute left-4 top-1/2 -translate-y-1/2 text-black/40">⌕</div>
        <input
          value={query}
          onChange={(e) => { setQuery(e.target.value); setIsOpen(true) }}
          onFocus={() => setIsOpen(true)}
          placeholder="Search country or city — e.g. Japan, Colombia, Sosua..."
          className="w-full pl-11 pr-4 py-4 bg-white border-2 border-black rounded-full text-[15px] focus:border-[#FF5A1F] focus:outline-none shadow-[4px_4px_0px_0px_black]"
        />
        {query && (
          <button onClick={() => setQuery("")} className="absolute right-4 top-1/2 -translate-y-1/2 w-7 h-7 bg-black text-white rounded-full text-xs">✕</button>
        )}
      </div>

      {/* Dropdown Results */}
      {isOpen && (
        <div className="absolute top-[60px] w-full bg-white border-2 border-black rounded-[20px] shadow-[6px_6px_0px_0px_black] overflow-hidden z-50">
          <div className="p-3 max-h-[360px] overflow-y-auto">
            <div className="text-[11px] uppercase tracking-widest opacity-40 px-3 py-2">
              {query ? `${filtered.length} destinations` : "Popular — 42 destinations live"}
            </div>
            {filtered.map((d) => (
              <button
                key={d.slug}
                onClick={() => goTo(d.slug)}
                className="w-full text-left px-3 py-3 rounded-xl hover:bg-black hover:text-white flex justify-between items-center group transition"
              >
                <div className="flex items-center gap-3">
                  <span className="text-xl">{d.flag}</span>
                  <div>
                    <div className="font-bold text-[15px]">{d.city}</div>
                    <div className="text-xs opacity-60 group-hover:opacity-80">{d.country}</div>
                  </div>
                </div>
                <div className="text-xs opacity-40 group-hover:opacity-100">→</div>
              </button>
            ))}
            {filtered.length === 0 && (
              <div className="px-3 py-8 text-center opacity-50 text-sm">
                No match. Try "Bali", "Mexico", or "Sosua"
              </div>
            )}
          </div>
          <div className="border-t border-black/10 p-3 bg-[#FFFBF6] flex justify-between items-center text-[11px]">
            <span className="opacity-60">Press Enter to search travelers in {query || "all hubs"}</span>
            <span className="font-bold">42 DESTINATIONS • VERIFIED</span>
          </div>
        </div>
      )}

      {/* Backdrop to close */}
      {isOpen && (
        <div className="fixed inset-0 z-40" onClick={() => setIsOpen(false)} />
      )}
    </div>
  )
}
