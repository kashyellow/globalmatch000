// components/MatchRevealCard.tsx - Blur-to-reveal card with Like → Match logic
// This replaces your static Sofia card - blur ON by default, OFF only on mutual like

"use client"
import { useState } from "react"

type Profile = {
  id: string
  displayName: string
  age: number
  bio: string
  travelStyles: string[]
  avatarUrl: string
  isVerified: boolean
  currentDestination: string
}

export default function MatchRevealCard({ profile, isInitiallyMatched = false }: { profile: Profile, isInitiallyMatched?: boolean }) {
  const [liked, setLiked] = useState(false)
  const [isMatch, setIsMatch] = useState(isInitiallyMatched)
  const [loading, setLoading] = useState(false)

  const handleLike = async () => {
    if (liked || loading) return
    setLoading(true)
    
    try {
      const res = await fetch("/api/like", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ toProfileId: profile.id })
      })
      const data = await res.json()
      
      setLiked(true)
      if (data.isMatch) {
        setIsMatch(true)
        // Trigger confetti / match modal
        console.log("MATCH!", profile.displayName)
      }
    } catch (e) {
      console.error(e)
    }
    setLoading(false)
  }

  const isBlurred = !isMatch // Blur ON by default, OFF only on mutual

  return (
    <div className="bg-white border-2 border-black rounded-[24px] overflow-hidden shadow-[4px_4px_0px_0px_black] hover:shadow-[6px_6px_0px_0px_black] transition">
      <div className="relative h-[320px] bg-gray-100">
        {/* Avatar - blurred unless matched */}
        {/* eslint-disable-next-line @next/next/no-img-element */}
        <img 
          src={profile.avatarUrl} 
          alt={profile.displayName}
          className={`w-full h-full object-cover transition-all duration-700 ${isBlurred ? "blur-[22px] scale-105" : "blur-0 scale-100"}`}
        />
        
        {/* Badges */}
        <div className="absolute top-3 left-3 flex gap-2">
          <div className={`px-2 py-1 rounded-full text-[10px] font-bold ${isBlurred ? "bg-black text-white" : "bg-green-500 text-white"}`}>
            {isBlurred ? "BLUR-TO-REVEAL ON" : "REVEALED ✓"}
          </div>
        </div>
        {profile.isVerified && (
          <div className="absolute top-3 right-3 px-2 py-1 bg-[#FF5A1F] text-white rounded-full text-[10px] font-bold">
            VERIFIED ✓
          </div>
        )}

        {/* Match overlay */}
        {isMatch && (
          <div className="absolute inset-0 bg-[#FF5A1F]/20 flex items-center justify-center">
            <div className="bg-white border-2 border-black rounded-full px-4 py-2 font-black shadow-[3px_3px_0px_0px_black] animate-bounce">
              IT'S A MATCH! 🎉
            </div>
          </div>
        )}

        {/* Bottom info */}
        <div className="absolute bottom-0 left-0 right-0 p-4 bg-gradient-to-t from-black/90 via-black/40 to-transparent text-white">
          <div className="font-bold text-lg flex items-center gap-2">
            {profile.displayName}, {profile.age} {profile.isVerified && "✓"}
            {isMatch && <span className="text-[#FF5A1F]">• Matched</span>}
          </div>
          <div className="flex gap-1 flex-wrap mt-1">
            {profile.travelStyles.map(s => (
              <span key={s} className="px-2 py-0.5 bg-white/20 backdrop-blur rounded-full text-[10px]">{s}</span>
            ))}
          </div>
          <div className="text-[11px] opacity-70 mt-1">{profile.currentDestination}</div>
        </div>
      </div>

      <div className="p-4">
        <p className="text-[13px] leading-snug line-clamp-3">{profile.bio}</p>
        
        <div className="flex gap-2 mt-4">
          {!isMatch ? (
            <>
              <button
                onClick={handleLike}
                disabled={liked || loading}
                className={`flex-1 py-2.5 rounded-full text-sm font-bold border-2 border-black transition ${
                  liked 
                    ? "bg-[#FFFBF6] text-black" 
                    : "bg-black text-white hover:bg-[#FF5A1F] hover:border-[#FF5A1F]"
                }`}
              >
                {loading ? "..." : liked ? "✓ Liked — waiting" : "♡ Like to reveal"}
              </button>
              <button className="w-11 h-11 border-2 border-black rounded-full hover:bg-black hover:text-white transition">×</button>
            </>
          ) : (
            <button className="flex-1 py-2.5 bg-[#FF5A1F] text-white rounded-full text-sm font-bold border-2 border-black shadow-[2px_2px_0px_0px_black]">
              💬 Message {profile.displayName}
            </button>
          )}
        </div>
        
        {liked && !isMatch && (
          <div className="text-[11px] opacity-50 text-center mt-2">
            Photo stays blurred until they like you back. Respect consent.
          </div>
        )}
      </div>
    </div>
  )
}
